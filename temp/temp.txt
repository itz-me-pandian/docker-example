Generate Pod YAML
kubectl run mypod --image=nginx --port=80 --dry-run=client -o yaml > pod.yaml

Generate Deployment YAML
kubectl create deployment myapp --image=nginx --replicas=3 --dry-run=client -o yaml > deployment.yaml

Generate Service YAML
ClusterIP:
kubectl create service clusterip myapp-service --tcp=80:80 --dry-run=client -o yaml > service.yaml

NodePort:
kubectl create service nodeport myapp-service --tcp=80:80 --node-port=30080 --dry-run=client -o yaml > service.yaml





kubectl create deployment frontend --image=nginx --replicas=2 --dry-run=client -o yaml > frontend-deployment.yaml
